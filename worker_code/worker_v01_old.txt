export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    const path = url.pathname;

    const NOTION_VERSION = "2022-06-28";
    const PROJECTS_DB = "3239f78fc0cf8014b2affdd98f184e2e";
    const DEVLOGS_DB = "3239f78fc0cf802fa392dd7241bc3141";
    const TASKS_DB = "3239f78f-c0cf-80b0-bcc3-e6f1ae2739f0";

    const DEVLOG_CONTAINS_SERIES_PROP = "Series";

    async function notionFetch(endpoint, options = {}) {
      const res = await fetch(`https://api.notion.com/v1/${endpoint}`, {
        ...options,
        headers: {
          "Authorization": `Bearer ${env.NOTION_KEY}`,
          "Notion-Version": NOTION_VERSION,
          "Content-Type": "application/json",
          ...(options.headers || {})
        }
      });
      return res.json();
    }

    async function getProjectMap() {
      const data = await notionFetch(`databases/${PROJECTS_DB}/query`, {
        method: "POST"
      });

      const map = new Map();
      (data.results || []).forEach(page => {
        const id = page.id.replace(/-/g, "");
        const name = page.properties.Name?.title?.[0]?.plain_text || "Untitled Project";
        map.set(id, name);
      });

      return map;
    }

    async function fetchBlocksRecursive(blockId) {
      let allBlocks = [];
      let cursor = undefined;

      while (true) {
        const params = cursor ? `?start_cursor=${cursor}` : "";
        const data = await notionFetch(`blocks/${blockId}/children${params}`, {
          method: "GET"
        });

        const results = data.results || [];
        for (const block of results) {
          if (block.has_children) {
            block.children = await fetchBlocksRecursive(block.id);
          } else {
            block.children = [];
          }
          allBlocks.push(block);
        }

        if (!data.has_more) break;
        cursor = data.next_cursor;
      }

      return allBlocks;
    }

    function isTextBlock(block) {
      return block.type === "paragraph" ||
             block.type === "heading_1" ||
             block.type === "heading_2" ||
             block.type === "heading_3" ||
             block.type === "bulleted_list_item" ||
             block.type === "numbered_list_item" ||
             block.type === "quote" ||
             block.type === "callout";
    }

    function getPlainTextFromRichText(rich) {
      return (rich || []).map(r => r.plain_text || "").join("");
    }

    function blockTextContains(block, marker) {
      if (!isTextBlock(block)) return false;
      const rich = block[block.type]?.rich_text || [];
      const text = getPlainTextFromRichText(rich);
      return text.includes(marker);
    }

    function isImageBlock(block) {
      return block.type === "image";
    }

    function getImageUrl(block) {
      if (!isImageBlock(block)) return "";
      const img = block.image;
      if (!img) return "";
      if (img.type === "file") return img.file?.url || "";
      if (img.type === "external") return img.external?.url || "";
      return "";
    }

    function getImageCaption(block) {
      if (!isImageBlock(block)) return "";
      const caption = block.image?.caption || [];
      return getPlainTextFromRichText(caption);
    }

    function extractSeriesImages(blocks, devlogMeta) {
      const seriesImages = [];
      let inSeriesRange = false;

      for (const block of blocks) {

        // RANGE START
        if (
          blockTextContains(block, "#SeriesStart") ||
          blockTextContains(block, "#Series--") // legacy
        ) {
          inSeriesRange = true;
          continue;
        }

        // RANGE END
        if (
          blockTextContains(block, "#SeriesEnd") ||
          blockTextContains(block, "#--Series") // legacy
        ) {
          inSeriesRange = false;
          continue;
        }

        // IMAGE INSIDE RANGE
        if (inSeriesRange && isImageBlock(block)) {
          const url = getImageUrl(block);
          if (!url) continue;

          seriesImages.push({
            url,
            caption: getImageCaption(block),
            devlogId: devlogMeta.id,
            devlogTitle: devlogMeta.title,
            devlogDate: devlogMeta.date || null,
            order: seriesImages.length
          });
          continue;
        }

        // SINGLE-IMAGE TAG
        if (isImageBlock(block)) {
          const caption = getImageCaption(block);
          if (caption && caption.includes("#Series")) {
            const url = getImageUrl(block);
            if (!url) continue;

            seriesImages.push({
              url,
              caption,
              devlogId: devlogMeta.id,
              devlogTitle: devlogMeta.title,
              devlogDate: devlogMeta.date || null,
              order: seriesImages.length
            });
          }
        }
      }

      return seriesImages;
    }

    // /devlogs-all
    if (path === "/devlogs-all") {
      const projectMap = await getProjectMap();

      const data = await notionFetch(`databases/${DEVLOGS_DB}/query`, {
        method: "POST"
      });

      const devlogs = (data.results || []).map(page => {
        const title = page.properties.Name?.title?.[0]?.plain_text || "Untitled Devlog";

        let thumbnail = "";
        const thumb = page.properties.Thumbnail?.files?.[0];
        if (thumb) thumbnail = thumb.file?.url || thumb.external?.url || "";

        const proj = page.properties.Project?.relation?.[0] || null;
        const projectId = proj ? proj.id.replace(/-/g, "") : null;
        const projectName = projectId ? projectMap.get(projectId) : null;

        const tags = page.properties.Tags?.multi_select?.map(t => t.name) || [];

        return {
          id: page.id,
          name: title,
          thumbnail,
          projectId,
          projectName,
          tags
        };
      });

      return new Response(JSON.stringify(devlogs), {
        headers: {
          "Content-Type": "application/json",
          "Access-Control-Allow-Origin": "*"
        }
      });
    }

    // /devlogs?projectId=XYZ
    if (path === "/devlogs") {
      const projectId = url.searchParams.get("projectId");
      if (!projectId) {
        return new Response(JSON.stringify({ error: "Missing projectId" }), {
          status: 400,
          headers: {
            "Content-Type": "application/json",
            "Access-Control-Allow-Origin": "*"
          }
        });
      }

      const cleanId = projectId.replace(/-/g, "");
      const projectMap = await getProjectMap();

      const data = await notionFetch(`databases/${DEVLOGS_DB}/query`, {
        method: "POST",
        body: JSON.stringify({
          filter: {
            property: "Project",
            relation: { contains: cleanId }
          }
        })
      });

      const devlogs = (data.results || []).map(page => {
        const title = page.properties.Name?.title?.[0]?.plain_text || "Untitled Devlog";

        let thumbnail = "";
        const thumb = page.properties.Thumbnail?.files?.[0];
        if (thumb) thumbnail = thumb.file?.url || thumb.external?.url || "";

        const tags = page.properties.Tags?.multi_select?.map(t => t.name) || [];

        return {
          id: page.id,
          name: title,
          thumbnail,
          projectId: cleanId,
          projectName: projectMap.get(cleanId),
          tags
        };
      });

      return new Response(JSON.stringify(devlogs), {
        headers: {
          "Content-Type": "application/json",
          "Access-Control-Allow-Origin": "*"
        }
      });
    }

    // /devlog?id=XYZ
    if (path === "/devlog") {
      const id = url.searchParams.get("id");
      if (!id) {
        return new Response(JSON.stringify({ error: "Missing id" }), {
          status: 400,
          headers: {
            "Content-Type": "application/json",
            "Access-Control-Allow-Origin": "*"
          }
        });
      }

      const projectMap = await getProjectMap();

      const pageData = await notionFetch(`pages/${id}`, { method: "GET" });
      const blocks = await fetchBlocksRecursive(id);

      const proj = pageData.properties.Project?.relation?.[0] || null;
      const projectId = proj ? proj.id.replace(/-/g, "") : null;
      const projectName = projectId ? projectMap.get(projectId) : null;

      const task = pageData.properties.Tasks?.relation?.[0] || null;
      const taskId = task ? task.id.replace(/-/g, "") : null;
      let taskName = null;
      if (task) {
        const taskPage = await notionFetch(`pages/${task.id}`, { method: "GET" });
        taskName = taskPage.properties.Name?.title?.[0]?.plain_text || null;
      }

      const title = pageData.properties.Name?.title?.[0]?.plain_text || "Untitled Devlog";
      const devlogDate =
        pageData.properties.Date?.date?.start ||
        pageData.created_time ||
        null;

      const containsSeries =
        pageData.properties[DEVLOG_CONTAINS_SERIES_PROP]?.checkbox === true;

      // ⭐ NEW: Thumbnail support for /devlog
      let thumbnail = "";
      const thumb = pageData.properties.Thumbnail?.files?.[0];
      if (thumb) thumbnail = thumb.file?.url || thumb.external?.url || "";

      let seriesImages = [];
      if (containsSeries) {
        seriesImages = extractSeriesImages(blocks, {
          id,
          title,
          date: devlogDate
        });
      }

      return new Response(JSON.stringify({
        id,
        title,
        projectId,
        projectName,
        taskId,
        taskName,
        blocks,
        containsSeries,
        seriesImages,
        thumbnail   // ⭐ NEW FIELD
      }), {
        headers: {
          "Content-Type": "application/json",
          "Access-Control-Allow-Origin": "*"
        }
      });
    }

    // /task?id=XYZ
    if (path === "/task") {
      const id = url.searchParams.get("id");
      if (!id) {
        return new Response(JSON.stringify({ error: "Missing id" }), {
          status: 400,
          headers: {
            "Content-Type": "application/json",
            "Access-Control-Allow-Origin": "*"
          }
        });
      }

      const pageData = await notionFetch(`pages/${id}`, { method: "GET" });
      const blocks = await fetchBlocksRecursive(id);

      const proj = pageData.properties.Projects?.relation?.[0] || null;
      const projectId = proj ? proj.id.replace(/-/g, "") : null;
      const projectName = projectId ? (await getProjectMap()).get(projectId) : null;

      const sessions = pageData.properties.Sessions?.relation || [];
      const devlogs = [];

      for (const session of sessions) {
        const devlogId = session.id;
        const devlogPage = await notionFetch(`pages/${devlogId}`, { method: "GET" });

        const title = devlogPage.properties.Name?.title?.[0]?.plain_text || "Untitled Devlog";

        let thumbnail = "";
        const thumb = devlogPage.properties.Thumbnail?.files?.[0];
        if (thumb) thumbnail = thumb.file?.url || thumb.external?.url || "";

        const devlogDate =
          devlogPage.properties.Date?.date?.start ||
          devlogPage.created_time ||
          null;

        const containsSeries =
          devlogPage.properties[DEVLOG_CONTAINS_SERIES_PROP]?.checkbox === true;

        let seriesImages = [];
        if (containsSeries) {
          const devlogBlocks = await fetchBlocksRecursive(devlogId);
          seriesImages = extractSeriesImages(devlogBlocks, {
            id: devlogId,
            title,
            date: devlogDate
          });
        }

        devlogs.push({
          id: devlogId,
          title,
          thumbnail,
          containsSeries,
          seriesImages
        });
      }

      return new Response(JSON.stringify({
        id,
        title: pageData.properties.Name?.title?.[0]?.plain_text || "Untitled Task",
        projectId,
        projectName,
        blocks,
        devlogs
      }), {
        headers: {
          "Content-Type": "application/json",
          "Access-Control-Allow-Origin": "*"
        }
      });
    }

    // DEFAULT: / → projects list
    const projectsData = await notionFetch(`databases/${PROJECTS_DB}/query`, {
      method: "POST"
    });

    const tasksData = await notionFetch(`databases/${TASKS_DB}/query`, {
      method: "POST"
    });

    const devlogsData = await notionFetch(`databases/${DEVLOGS_DB}/query`, {
      method: "POST"
    });

    const tasksByProject = {};
    const devlogsByTask = {};

    for (const d of (devlogsData.results || [])) {
      const devlogId = d.id;
      const title = d.properties.Name?.title?.[0]?.plain_text || "Untitled Devlog";

      let thumbnail = "";
      const thumb = d.properties.Thumbnail?.files?.[0];
      if (thumb) thumbnail = thumb.file?.url || thumb.external?.url || "";

      const taskRels = d.properties.Tasks?.relation || [];
      for (const t of taskRels) {
        const taskId = t.id.replace(/-/g, "");
        if (!devlogsByTask[taskId]) devlogsByTask[taskId] = [];
        devlogsByTask[taskId].push({
          id: devlogId,
          title,
          thumbnail
        });
      }
    }

    for (const t of (tasksData.results || [])) {
      const taskId = t.id.replace(/-/g, "");
      const taskName = t.properties.Name?.title?.[0]?.plain_text || "Untitled Task";

      const projRels = t.properties.Projects?.relation || [];
      for (const p of projRels) {
        const projectId = p.id.replace(/-/g, "");
        if (!tasksByProject[projectId]) tasksByProject[projectId] = [];
        tasksByProject[projectId].push({
          id: taskId,
          name: taskName,
          devlogs: devlogsByTask[taskId] || []
        });
      }
    }

    const projects = (projectsData.results || []).map(page => {
      const projectId = page.id.replace(/-/g, "");
      const title = page.properties.Name?.title?.[0]?.plain_text || "Untitled Project";
      const description = page.properties.Description?.rich_text?.[0]?.plain_text || "";

      let thumbnail = "";
      const thumb = page.properties.Thumbnail?.files?.[0];
      if (thumb) thumbnail = thumb.file?.url || thumb.external?.url || "";

      return {
        id: page.id,
        name: title,
        description,
        thumbnail,
        tasks: tasksByProject[projectId] || []
      };
    });

    return new Response(JSON.stringify(projects), {
      headers: {
        "Content-Type": "application/json",
        "Access-Control-Allow-Origin": "*"
      }
    });
  }
};
